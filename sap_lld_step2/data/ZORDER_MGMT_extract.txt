REPORT zrp_order_batch_run.

TABLES: zorder_header.

TYPES: BEGIN OF ty_run_stat,
         processed TYPE i,
         failed    TYPE i,
         skipped   TYPE i,
       END OF ty_run_stat.

DATA: gt_orders  TYPE STANDARD TABLE OF zorder_header,
      gs_stat    TYPE ty_run_stat,
      go_proc    TYPE REF TO zcl_order_processor,
      gv_runid   TYPE char20.

SELECTION-SCREEN BEGIN OF BLOCK b1 WITH FRAME TITLE TEXT-001.
  SELECT-OPTIONS: s_vbeln FOR zorder_header-vbeln,
                  s_erdat FOR zorder_header-erdat.
  PARAMETERS: p_test AS CHECKBOX DEFAULT 'X',
              p_maxrec TYPE i DEFAULT 500.
SELECTION-SCREEN END OF BLOCK b1.

START-OF-SELECTION.
  PERFORM get_run_id.
  PERFORM select_orders.
  PERFORM process_orders.
  PERFORM write_summary.

FORM get_run_id.
  CALL FUNCTION 'ZFM_ORDER_NUMBER_RANGE'
    EXPORTING
      iv_object    = 'ZBATCHRUN'
    IMPORTING
      ev_number    = gv_runid
    EXCEPTIONS
      no_range     = 1
      OTHERS       = 2.

  IF sy-subrc <> 0.
    gv_runid = |FALLBACK{ sy-datum }{ sy-uzeit }|.
  ENDIF.
ENDFORM.

FORM select_orders.
  SELECT * FROM zorder_header
    INTO TABLE gt_orders
    UP TO p_maxrec ROWS
    WHERE vbeln IN s_vbeln
      AND erdat IN s_erdat
      AND status = 'NEW'.

  IF sy-subrc <> 0.
    MESSAGE 'No orders found for selection' TYPE 'S'.
    RETURN.
  ENDIF.
ENDFORM.

FORM process_orders.
  DATA: lv_ok TYPE abap_bool.

  CREATE OBJECT go_proc.

  LOOP AT gt_orders INTO DATA(ls_order).
    IF p_test = abap_true.
      ADD 1 TO gs_stat-skipped.
      CONTINUE.
    ENDIF.

    lv_ok = go_proc->process_single_order( is_header = ls_order ).

    IF lv_ok = abap_true.
      ADD 1 TO gs_stat-processed.
    ELSE.
      ADD 1 TO gs_stat-failed.
      zcl_order_logger=>log_error(
        iv_object = 'ZRP_ORDER_BATCH_RUN'
        iv_key    = ls_order-vbeln
        iv_text   = 'Processing returned false' ).
    ENDIF.
  ENDLOOP.
ENDFORM.

FORM write_summary.
  WRITE: / 'Run ID:', gv_runid.
  WRITE: / 'Processed:', gs_stat-processed.
  WRITE: / 'Failed:', gs_stat-failed.
  WRITE: / 'Skipped:', gs_stat-skipped.

  zcl_order_logger=>log_info(
    iv_object = 'ZRP_ORDER_BATCH_RUN'
    iv_key    = gv_runid
    iv_text   = |Batch complete: { gs_stat-processed } ok| ).
ENDFORM.

CLASS zcl_order_processor DEFINITION PUBLIC CREATE PUBLIC.

  PUBLIC SECTION.

CLASS zcl_order_processor IMPLEMENTATION.

  METHOD init_dependencies.
    IF mo_validator IS INITIAL.
      CREATE OBJECT mo_validator.
      CREATE OBJECT mo_pricing.
      CREATE OBJECT mo_stock.
      CREATE OBJECT mo_persist.
    ENDIF.
  ENDMETHOD.

  METHOD build_item_list.
    SELECT * FROM zorder_item
      INTO TABLE rt_items
      WHERE vbeln = iv_vbeln
      ORDER BY posnr ASCENDING.
  ENDMETHOD.

  METHOD process_single_order.
    DATA: lt_items TYPE ztt_order_item,
          lv_total TYPE netwr.

    init_dependencies( ).
    rv_ok = abap_false.

    IF mo_validator->validate_header( is_header ) = abap_false.
      zcl_order_logger=>log_error(
        iv_object = 'ZCL_ORDER_PROCESSOR'
        iv_key    = is_header-vbeln
        iv_text   = 'Header validation failed' ).
      RETURN.
    ENDIF.

    lt_items = build_item_list( is_header-vbeln ).

    IF lines( lt_items ) = 0.
      zcl_order_logger=>log_error(
        iv_object = 'ZCL_ORDER_PROCESSOR'
        iv_key    = is_header-vbeln
        iv_text   = 'Order has no items' ).
      RETURN.
    ENDIF.

    IF mo_stock->reserve_for_order(
         iv_vbeln = is_header-vbeln
         it_items = lt_items ) = abap_false.
      RETURN.
    ENDIF.

    lv_total = mo_pricing->calculate_order_total(
                 is_header = is_header
                 it_items  = lt_items ).

    mo_persist->save_order_total(
      iv_vbeln = is_header-vbeln
      iv_total = lv_total ).

    zcl_notification_sender=>notify_order_confirmed(
      iv_vbeln = is_header-vbeln
      iv_kunnr = is_header-kunnr ).

    rv_ok = abap_true.
  ENDMETHOD.

  METHOD reprocess_failed.
    DATA: ls_header TYPE zorder_header.

    SELECT SINGLE * FROM zorder_header
      INTO ls_header
      WHERE vbeln = iv_vbeln.

    IF sy-subrc <> 0.
      rv_ok = abap_false.
      RETURN.
    ENDIF.

    rv_ok = process_single_order( ls_header ).
  ENDMETHOD.

ENDCLASS.

             validate_customer

               RETURNING VALUE(rv_ok) TYPE abap_bool,

               IMPORTING iv_kunnr     TYPE kunnr
                         iv_amount    TYPE netwr
               RETURNING VALUE(rv_ok) TYPE abap_bool.

ENDCLASS.

CLASS zcl_order_validator IMPLEMENTATION.

  METHOD validate_header.
    rv_ok = abap_false.

    IF is_header-vbeln IS INITIAL.
      RETURN.
    ENDIF.

    IF is_header-kunnr IS INITIAL.
      zcl_order_logger=>log_error(
        iv_object = 'ZCL_ORDER_VALIDATOR'
        iv_key    = is_header-vbeln
        iv_text   = 'Customer missing on header' ).
      RETURN.
    ENDIF.

    IF validate_customer( is_header-kunnr ) = abap_false.
      RETURN.
    ENDIF.

    IF is_header-netwr > 0.
      IF check_credit_limit(
           iv_kunnr  = is_header-kunnr
           iv_amount = is_header-netwr ) = abap_false.
        RETURN.
      ENDIF.
    ENDIF.

    rv_ok = abap_true.
  ENDMETHOD.

  METHOD validate_customer.
    DATA: lv_blocked TYPE abap_bool.

    SELECT SINGLE blocked FROM zcust_master
      INTO lv_blocked
      WHERE kunnr = iv_kunnr.

    IF sy-subrc <> 0.
      rv_ok = abap_false.
      RETURN.
    ENDIF.

    IF lv_blocked = abap_true.
      zcl_order_logger=>log_error(
        iv_object = 'ZCL_ORDER_VALIDATOR'
        iv_key    = iv_kunnr
        iv_text   = 'Customer is blocked' ).
      rv_ok = abap_false.
      RETURN.
    ENDIF.

    rv_ok = abap_true.
  ENDMETHOD.

  METHOD check_credit_limit.
    DATA: lv_available TYPE netwr.

    CALL FUNCTION 'ZFM_CUSTOMER_CREDIT_CHECK'
      EXPORTING
        iv_kunnr     = iv_kunnr
      IMPORTING
        ev_available = lv_available
      EXCEPTIONS
        no_limit_set = 1
        OTHERS       = 2.

    IF sy-subrc <> 0.
      rv_ok = abap_false.
      RETURN.
    ENDIF.

    IF lv_available >= iv_amount.
      rv_ok = abap_true.
    ELSE.
      rv_ok = abap_false.
    ENDIF.
  ENDMETHOD.

ENDCLASS.

CLASS zcl_pricing_engine DEFINITION PUBLIC CREATE PUBLIC.

  PUBLIC SECTION.

               IMPORTING is_header       TYPE zorder_header
                         it_items        TYPE ztt_order_item
               RETURNING VALUE(rv_total) TYPE netwr,
             get_base_price
               IMPORTING iv_matnr        TYPE matnr
                         iv_kunnr        TYPE kunnr
               RETURNING VALUE(rv_price) TYPE netwr.

  PRIVATE SECTION.
    DATA: mo_discount TYPE REF TO zcl_discount_calculator,
          mo_tax      TYPE REF TO zcl_tax_calculator,
          mo_shipping TYPE REF TO zcl_shipping_calculator.

ENDCLASS.

CLASS zcl_pricing_engine IMPLEMENTATION.

  METHOD get_base_price.
    SELECT SINGLE kbetr FROM zprice_cond
      INTO rv_price
      WHERE matnr = iv_matnr
        AND kunnr = iv_kunnr
        AND datbi >= sy-datum
        AND datab <= sy-datum.

    IF sy-subrc <> 0.
      SELECT SINGLE kbetr FROM zprice_cond
        INTO rv_price
        WHERE matnr = iv_matnr
          AND kunnr = space
          AND datbi >= sy-datum.
    ENDIF.
  ENDMETHOD.

  METHOD calculate_order_total.
    DATA: lv_gross    TYPE netwr,
          lv_discount TYPE netwr,
          lv_tax      TYPE netwr,
          lv_ship     TYPE netwr,
          lv_line     TYPE netwr.

    IF mo_discount IS INITIAL.
      CREATE OBJECT mo_discount.
      CREATE OBJECT mo_tax.
      CREATE OBJECT mo_shipping.
    ENDIF.

    LOOP AT it_items INTO DATA(ls_item).
      lv_line = get_base_price(
                  iv_matnr = ls_item-matnr
                  iv_kunnr = is_header-kunnr ) * ls_item-kwmeng.
      ADD lv_line TO lv_gross.
    ENDLOOP.

    lv_discount = mo_discount->calculate_discount(
                    iv_kunnr = is_header-kunnr
                    iv_gross = lv_gross ).

    lv_tax = mo_tax->calculate_tax(
               iv_land1  = is_header-land1
               iv_amount = lv_gross - lv_discount ).

    lv_ship = mo_shipping->calculate_shipping(
                iv_kunnr = is_header-kunnr
                iv_items = lines( it_items ) ).

    rv_total = lv_gross - lv_discount + lv_tax + lv_ship.
  ENDMETHOD.

ENDCLASS.

  PUBLIC SECTION.

               IMPORTING iv_kunnr       TYPE kunnr

CLASS zcl_discount_calculator IMPLEMENTATION.

  METHOD get_customer_tier.
    SELECT SINGLE tier FROM zcust_master
      INTO rv_tier
      WHERE kunnr = iv_kunnr.

    IF sy-subrc <> 0.
      rv_tier = 'ST'.
    ENDIF.
  ENDMETHOD.

  METHOD calculate_discount.
    DATA: lv_tier    TYPE char2,
          lv_percent TYPE p DECIMALS 2.

    lv_tier = get_customer_tier( iv_kunnr ).

    SELECT SINGLE percent FROM zdiscount_rule
      INTO lv_percent
      WHERE tier      = lv_tier
        AND min_value <= iv_gross
        AND valid_to  >= sy-datum.

    IF sy-subrc <> 0.
      rv_disc = 0.
      zcl_order_logger=>log_info(
        iv_object = 'ZCL_DISCOUNT_CALCULATOR'
        iv_key    = iv_kunnr
        iv_text   = |No discount rule for tier { lv_tier }| ).
      RETURN.
    ENDIF.

    rv_disc = iv_gross * lv_percent / 100.

    IF rv_disc > iv_gross / 2.
      rv_disc = iv_gross / 2.
      zcl_order_logger=>log_info(
        iv_object = 'ZCL_DISCOUNT_CALCULATOR'
        iv_key    = iv_kunnr
        iv_text   = 'Discount capped at 50 percent' ).
    ENDIF.
  ENDMETHOD.

ENDCLASS.

               IMPORTING iv_vbeln     TYPE vbeln
               RETURNING VALUE(rv_ok) TYPE abap_bool.

CLASS zcl_stock_manager IMPLEMENTATION.

  METHOD reserve_for_order.
    DATA: lv_avail TYPE kwmeng.

    rv_ok = abap_true.

    LOOP AT it_items INTO DATA(ls_item).
      CALL FUNCTION 'ZFM_STOCK_CHECK'
        EXPORTING
          iv_matnr   = ls_item-matnr
          iv_werks   = ls_item-werks
        IMPORTING
          ev_avail   = lv_avail
        EXCEPTIONS
          not_found  = 1
          OTHERS     = 2.

      IF sy-subrc <> 0 OR lv_avail < ls_item-kwmeng.
        zcl_order_logger=>log_error(
          iv_object = 'ZCL_STOCK_MANAGER'
          iv_key    = iv_vbeln
          iv_text   = |Insufficient stock for { ls_item-matnr }| ).
        rv_ok = abap_false.
        RETURN.
      ENDIF.
    ENDLOOP.

    LOOP AT it_items INTO ls_item.
      UPDATE zstock_level
        SET reserved = reserved + ls_item-kwmeng
        WHERE matnr = ls_item-matnr
          AND werks = ls_item-werks.
    ENDLOOP.

    COMMIT WORK.
  ENDMETHOD.

  METHOD release_reservation.
    DATA: lt_items TYPE ztt_order_item.

    SELECT * FROM zorder_item
      INTO TABLE lt_items
      WHERE vbeln = iv_vbeln.

    LOOP AT lt_items INTO DATA(ls_item).
      UPDATE zstock_level
        SET reserved = reserved - ls_item-kwmeng
        WHERE matnr = ls_item-matnr
          AND werks = ls_item-werks.
    ENDLOOP.

    COMMIT WORK.
    rv_ok = abap_true.
  ENDMETHOD.

ENDCLASS.

  METHOD update_status.
    UPDATE zorder_header
      SET status = iv_status
      WHERE vbeln = iv_vbeln.

    IF sy-subrc = 0.
      INSERT INTO zorder_status VALUES @( VALUE #(
        vbeln  = iv_vbeln
        status = iv_status
        chdat  = sy-datum
        chusr  = sy-uname ) ).
      COMMIT WORK.
      rv_ok = abap_true.
    ELSE.
      rv_ok = abap_false.
    ENDIF.
  ENDMETHOD.

    ENDIF.

    ls_header-vbeln = rv_vbeln.
    ls_header-erdat = sy-datum.
    ls_header-status = 'NEW'.

    INSERT zorder_header FROM ls_header.
    COMMIT WORK.
  ENDMETHOD.

FUNCTION zfm_order_number_range.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_OBJECT) TYPE  CHAR10
*"  EXPORTING
*"     VALUE(EV_NUMBER) TYPE  CHAR20
*"  EXCEPTIONS
*"      NO_RANGE
*"----------------------------------------------------------------------

  DATA: lv_nr     TYPE char20,
        lv_nrnr   TYPE nrnr.

  CLEAR ev_number.

  CASE iv_object.
    WHEN 'ZORDER'.
      lv_nrnr = '01'.
    WHEN 'ZBATCHRUN'.
      lv_nrnr = '02'.
    WHEN OTHERS.
      RAISE no_range.
  ENDCASE.

  CALL FUNCTION 'NUMBER_GET_NEXT'
    EXPORTING
      nr_range_nr = lv_nrnr
      object      = 'ZORDERNR'
    IMPORTING
      number      = lv_nr
    EXCEPTIONS
      OTHERS      = 1.

  IF sy-subrc <> 0.
    RAISE no_range.
  ENDIF.

  ev_number = lv_nr.

ENDFUNCTION.

  METHOD notify_order_confirmed.
    DATA: lv_mail TYPE ad_smtpadr.

    SELECT SINGLE smtp_addr FROM zcust_master
      INTO lv_mail
      WHERE kunnr = iv_kunnr.

    IF sy-subrc <> 0 OR lv_mail IS INITIAL.
      zcl_order_logger=>log_info(
        iv_object = 'ZCL_NOTIFICATION_SENDER'
        iv_key    = iv_kunnr
        iv_text   = 'No email address on customer' ).

      vbeln    = iv_vbeln

    INSERT INTO znotif_queue VALUES @( VALUE #(
      vbeln    = iv_vbeln
      recip    = 'orders-support@example.com'
      msgtype  = 'FAIL'
      msgtext  = iv_text
      crdat    = sy-datum
      sent     = abap_false ) ).

    COMMIT WORK.
  ENDMETHOD.

  METHOD flush_queue.
    DATA: lt_queue TYPE STANDARD TABLE OF znotif_queue.

    SELECT * FROM znotif_queue
      INTO TABLE lt_queue
      WHERE sent = abap_false.

        SET sent = abap_true
        WHERE vbeln   = ls_q-vbeln
          AND msgtype = ls_q-msgtype.
      ADD 1 TO rv_sent.

  ENDMETHOD.

REPORT zrp_order_summary_report.

TYPES: BEGIN OF ty_summary,
         kunnr    TYPE kunnr,
         name1    TYPE name1,
         cnt      TYPE i,
         total    TYPE netwr,
       END OF ty_summary.

DATA: gt_summary TYPE STANDARD TABLE OF ty_summary,
      go_persist TYPE REF TO zcl_order_persistence,
      go_alv     TYPE REF TO cl_salv_table.

SELECTION-SCREEN BEGIN OF BLOCK b1 WITH FRAME.
  SELECT-OPTIONS: s_kunnr FOR gt_summary-kunnr,
                  s_datum FOR sy-datum.
  PARAMETERS: p_closed AS CHECKBOX.
SELECTION-SCREEN END OF BLOCK b1.

START-OF-SELECTION.
  PERFORM collect_data.
  PERFORM enrich_names.
  PERFORM display_alv.

FORM collect_data.
  DATA: lv_status TYPE char4.

  IF p_closed = abap_true.
    lv_status = 'CLSD'.
  ELSE.
    lv_status = 'NEW'.
  ENDIF.

  SELECT kunnr COUNT( * ) AS cnt SUM( netwr ) AS total
    FROM zorder_header
    INTO CORRESPONDING FIELDS OF TABLE gt_summary
    WHERE kunnr  IN s_kunnr
      AND erdat  IN s_datum
      AND status = lv_status
    GROUP BY kunnr.

  IF sy-subrc <> 0.
    MESSAGE 'No data for selection' TYPE 'S' DISPLAY LIKE 'W'.
    LEAVE LIST-PROCESSING.
  ENDIF.
ENDFORM.

FORM enrich_names.
  LOOP AT gt_summary ASSIGNING FIELD-SYMBOL(<fs>).
    SELECT SINGLE name1 FROM zcust_master
      INTO <fs>-name1
      WHERE kunnr = <fs>-kunnr.
  ENDLOOP.
ENDFORM.

FORM display_alv.
  TRY.
      cl_salv_table=>factory(
        IMPORTING r_salv_table = go_alv
        CHANGING  t_table      = gt_summary ).

      go_alv->display( ).

    CATCH cx_salv_msg.
      zcl_order_logger=>log_error(
        iv_object = 'ZRP_ORDER_SUMMARY_REPORT'
        iv_key    = sy-uname
        iv_text   = 'ALV display failed' ).
  ENDTRY.
ENDFORM.
