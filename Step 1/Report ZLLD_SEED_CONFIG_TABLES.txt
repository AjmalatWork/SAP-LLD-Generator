REPORT zlld_seed_config_tables.

*&---------------------------------------------------------------------*
*& ZLLD_SEED_CONFIG_TABLES
*&---------------------------------------------------------------------*
*  Seeds ZDCCRT_EKIND_MAP and ZDCCRT_CONFIG, the two config tables used
*  by ZLLD_EXTRACT_TO_DB (edge-kind derivation) and ZLLD_GET_CANDIDATES
*  (scoring weights, score thresholds, tier multipliers). Run once
*  after creating the tables in SE11, before the first extraction or
*  retrieval run. Uses MODIFY throughout, so re-running it later (e.g.
*  after editing a VALUE #( ... ) list below) is always safe.
*
*  ZDCCRT_EKIND_MAP is a table rather than a hardcoded CASE - see
*  ZLLD_EXTRACT_TO_DB's ZLLD_LOOKUP_EDGE_KIND form. Adding a row here
*  for a newly-observed DEPENDENCY_TYPE never needs a transport, only a
*  data change (SM30 on ZDCCRT_EKIND_MAP directly also works going
*  forward - this report is just the bulk one-time load).
*
*  CALL_LIKE is true only for METH/OM (calls_method) and FUNC
*  (calls_function) - every other DEPENDENCY_TYPE is a structural or
*  reference edge, not a call.
*&---------------------------------------------------------------------*

START-OF-SELECTION.
  PERFORM seed_edge_kind_map.
  PERFORM seed_config.
  WRITE: / 'ZDCCRT_EKIND_MAP and ZDCCRT_CONFIG seeded (or re-seeded, unchanged if already current).'.

*&---------------------------------------------------------------------*
*&      Form  SEED_EDGE_KIND_MAP
*&---------------------------------------------------------------------*
FORM seed_edge_kind_map.
  TYPES: tt_edge_seed TYPE STANDARD TABLE OF zdccrt_ekind_map WITH EMPTY KEY.

  DATA(lt_seed) = VALUE tt_edge_seed(
    ( dependency_type = 'METH' edge_kind = 'calls_method'        is_call_like = abap_true  )
    ( dependency_type = 'OM'   edge_kind = 'calls_method'        is_call_like = abap_true  )
    ( dependency_type = 'FUNC' edge_kind = 'calls_function'      is_call_like = abap_true  )
    ( dependency_type = 'CLAS' edge_kind = 'class_ref'           is_call_like = abap_false )
    ( dependency_type = 'STRU' edge_kind = 'structure_ref'       is_call_like = abap_false )
    ( dependency_type = 'MESS' edge_kind = 'message_ref'         is_call_like = abap_false )
    ( dependency_type = 'OA'   edge_kind = 'reference'           is_call_like = abap_false )
    ( dependency_type = 'DTEL' edge_kind = 'data_element_ref'    is_call_like = abap_false )
    ( dependency_type = 'INCL' edge_kind = 'includes'            is_call_like = abap_false )
    ( dependency_type = 'INTF' edge_kind = 'implements'          is_call_like = abap_false )
    ( dependency_type = 'DGT'  edge_kind = 'generic_type_ref'    is_call_like = abap_false )
    ( dependency_type = 'FUGR' edge_kind = 'function_group_ref'  is_call_like = abap_false )
    ( dependency_type = 'PROG' edge_kind = 'program_ref'         is_call_like = abap_false )
    ( dependency_type = 'TABL' edge_kind = 'table_ref'           is_call_like = abap_false )
    ( dependency_type = 'MSAG' edge_kind = 'message_class_ref'   is_call_like = abap_false )
    ( dependency_type = 'TTYP' edge_kind = 'table_type_ref'      is_call_like = abap_false )
    ( dependency_type = 'TYPE' edge_kind = 'type_ref'            is_call_like = abap_false )
    ( dependency_type = 'TRAN' edge_kind = 'transaction_ref'     is_call_like = abap_false )
  ).

  LOOP AT lt_seed INTO DATA(ls_seed).
    MODIFY zdccrt_ekind_map FROM ls_seed.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  SEED_CONFIG
*&---------------------------------------------------------------------*
*       STOPWORDS is a short list of ABAP-generic noise tokens filtered
*       out before tokenizing, so they don't dominate every chunk
*       indiscriminately. Editable directly via SM30 on ZDCCRT_CONFIG;
*       this report only sets the initial baseline.
*
*       CHUNK_MERGE_MIN_LINES is the minimum chunk size, in lines - a
*       fragment shorter than this gets merged into a neighboring
*       chunk instead of standing alone.
*
*       EXTRACT_COMMIT_BATCH_SIZE is how many objects EXTRACT_TO_DB
*       processes between COMMIT WORKs during a run - keeps one very
*       large package (thousands of objects) from running as a single
*       giant transaction. Lower it if a run is hitting lock/rollback-
*       segment pressure; raise it (or set to 0 to disable mid-run
*       commits entirely) if a package is small enough that the default
*       is just adding unnecessary commit overhead.
*
*       SEMANTIC_WEIGHT/STRUCTURAL_WEIGHT control how much each score
*       component contributes to an object's base score. SEMANTIC_WEIGHT
*       starts low: lexical text similarity alone is a weaker signal
*       than graph proximity, so it is weighted accordingly.
*
*       SEED_MIN_SEMANTIC_SCORE, NEW_OBJECT_SCORE_FLOOR, and
*       NEW_OBJECT_SPREAD_THRESHOLD are score thresholds tuned to this
*       scoring scheme's actual value range; revisit them as usage
*       confirms or adjusts what "strong" and "weak" mean in practice.
*
*       MIN_CHUNK_LENGTH is the minimum chunk character length to count
*       as a scoreable "document" at all - read by both
*       ZLLD_GET_CANDIDATES (query time) and ZLLD_EXTRACT_TO_DB (when
*       precomputing a chunk's own CHUNK_NORM), which is why it lives
*       here as one shared value rather than a constant maintained
*       separately in each program.
*
*       TOP_SEED_OBJECTS, STRUCTURAL_SEED_SCORE, STRUCTURAL_HOP1_SCORE,
*       STRUCTURAL_HOP2_SCORE, STRUCTURAL_SHARED_TABLE_BONUS,
*       CERTAIN_HOP_RADIUS, and CERTAIN_CONFLICT_MARGIN drive
*       ZLLD_GET_CANDIDATES' structural-scoring and CERTAIN-tier
*       disagreement logic - see COMPUTE_SEED_STRUCT_SCORES,
*       APPLY_TIER_ADJUSTMENT, and DETECT_CERTAIN_HINT_CONFLICT there
*       for exactly how each one is used.
*&---------------------------------------------------------------------*
FORM seed_config.
  TYPES: tt_config_seed TYPE STANDARD TABLE OF zdccrt_config WITH EMPTY KEY.

  DATA(lt_seed) = VALUE tt_config_seed(
    ( config_key = 'STOPWORDS'
      config_value = 'data,type,types,begin,end,if,endif,else,elseif,endcase,case,when,'
                   && 'value,values,class,endclass,method,endmethod,form,endform,perform,'
                   && 'function,endfunction,loop,endloop,into,from,where,select,table,'
                   && 'standard,structure,changing,using,importing,exporting,returning,'
                   && 'optional,is,not,and,or,check,clear,move,append,read,write,'
                   && 'ev,lv,iv,cv,gv,gt,lt,ls,gs,rt,ct,it,wa,fs,'
                   && 'ew,lw,iw,cw,gw,gi,li,ri,ci,ii,'
                   && 'eq,ei,er,et,es,rs,ne,ge,le,co,cn,ca,na,cs,ns,cp,np,bt,nb,'
                   && 'try,catch,endtry,cleanup,raise,exception,exceptions,do,enddo,'
                   && 'while,endwhile,interface,endinterface,definition,implementation,'
                   && 'public,private,protected,section,constants,statics,initial,'
                   && 'space,abap' )
    ( config_key = 'CHUNK_MERGE_MIN_LINES' config_value = '3' )
    ( config_key = 'EXTRACT_COMMIT_BATCH_SIZE' config_value = '200' )
    ( config_key = 'MIN_CHUNK_LENGTH' config_value = '15' )
    ( config_key = 'SEMANTIC_WEIGHT'   config_value = '0.30' )
    ( config_key = 'STRUCTURAL_WEIGHT' config_value = '0.70' )
    ( config_key = 'SEED_MIN_SEMANTIC_SCORE'      config_value = '0.30' )
    ( config_key = 'NEW_OBJECT_SCORE_FLOOR'       config_value = '0.20' )
    ( config_key = 'NEW_OBJECT_SPREAD_THRESHOLD'  config_value = '0.05' )
    ( config_key = 'LIKELY_NAMED_MULTIPLIER' config_value = '1.5' )
    ( config_key = 'LIKELY_HOP1_MULTIPLIER'  config_value = '1.2' )
    ( config_key = 'UNSURE_NAMED_MULTIPLIER' config_value = '1.1' )
    ( config_key = 'TOP_SEED_OBJECTS'         config_value = '3' )
    ( config_key = 'STRUCTURAL_SEED_SCORE'    config_value = '1.0' )
    ( config_key = 'STRUCTURAL_HOP1_SCORE'    config_value = '0.6' )
    ( config_key = 'STRUCTURAL_HOP2_SCORE'    config_value = '0.3' )
    ( config_key = 'STRUCTURAL_SHARED_TABLE_BONUS' config_value = '0.4' )
    ( config_key = 'CERTAIN_HOP_RADIUS'       config_value = '1' )
    ( config_key = 'CERTAIN_CONFLICT_MARGIN'  config_value = '0.15' )
  ).

  LOOP AT lt_seed INTO DATA(ls_seed).
    MODIFY zdccrt_config FROM ls_seed.
  ENDLOOP.
ENDFORM.
