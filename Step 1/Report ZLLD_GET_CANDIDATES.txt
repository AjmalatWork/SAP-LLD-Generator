REPORT zlld_get_candidates.

*&---------------------------------------------------------------------*
*& ZLLD_GET_CANDIDATES
*&---------------------------------------------------------------------*
*  The report a developer runs to find which existing ABAP objects a
*  new requirement most likely belongs to. Searches the query-ready
*  tables ZLLD_EXTRACT_TO_DB builds (ZDCCRT_OBJECTS, ZDCCRT_OBJ_CALLS,
*  ZDCCRT_OBJ_DDREF, ZDCCRT_DDIC_OBJ, ZDCCRT_CHUNKS, ZDCCRT_CHUNK_TOK,
*  ZDCCRT_EKIND_MAP), scores and ranks candidate objects, detects
*  disagreements between a developer's own scoping note and the
*  independent ranking, flags likely-new-work requirements, and writes
*  a candidate file in the format the downstream gem tool reads.
*
*  READ-ONLY against those tables - never writes to ZDCCRT_OBJECTS/
*  ZDCCRT_OBJ_CALLS/etc.
*
*  Semantic scoring is TF-IDF cosine similarity over sparse lexical
*  vectors - see COMPUTE_IDF_TABLE/COMPUTE_SEMANTIC_SCORES below. It is
*  a cosine similarity bounded [0,1] over non-negative-weighted vectors,
*  which is what the score thresholds below assume; a raw summed TF-IDF
*  score would not stay comparable across queries the same way.
*
*  Every query-time table access is scoped to the requirement's own
*  vocabulary (WHERE TOKEN IN <requirement's distinct tokens>), never to
*  the whole selected package(s) - cost tracks requirement length, not
*  corpus size, since a package can hold thousands of objects. What that
*  scoping can't reach (a chunk's own full-vocabulary vector norm) is
*  precomputed once at extraction time instead - see
*  ZLLD_COMPUTE_CHUNK_NORM in ZLLD_EXTRACT_TO_DB_F01 and
*  ZDCCRT_CHUNKS-CHUNK_NORM/CHUNK_LENGTH.
*
*  ZDCCRT_CONFIG's SEED_MIN_SEMANTIC_SCORE, NEW_OBJECT_SCORE_FLOOR, and
*  NEW_OBJECT_SPREAD_THRESHOLD are tunable via SM30 without a transport;
*  treat them as working values to revisit as usage confirms or adjusts
*  them, not as fixed constants.
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*

  ls_value-key = 'UNSURE'.  ls_value-text = 'UNSURE'.  APPEND ls_value TO lt_values.

  " (ID TYPE VRM_ID, VALUES TYPE VRM_VALUES) - no TABLES parameters at
  " all, unlike the classic TABLES-based signature.

    EXPORTING
      id     = 'P_CONF'
      values = lt_values.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  VALIDATE_INPUTS
*&---------------------------------------------------------------------*
*       A scoping note requires a tier and vice versa. P_CONF is
*       normalized to uppercase; blank means no tier.
*&---------------------------------------------------------------------*
FORM validate_inputs.
  CLEAR gv_error.
  p_conf = to_upper( p_conf ).

  IF p_note IS NOT INITIAL AND p_conf IS INITIAL.
    MESSAGE 'CONFIDENCE is required when a scoping note is given.' TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
    RETURN.
  ENDIF.

  IF p_note IS INITIAL AND p_conf IS NOT INITIAL.
    MESSAGE 'CONFIDENCE must be blank when no scoping note is given.' TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
    RETURN.
  ENDIF.

  IF p_conf IS NOT INITIAL
     AND p_conf <> 'UNSURE' AND p_conf <> 'LIKELY' AND p_conf <> 'CERTAIN'.
    MESSAGE |CONFIDENCE must be UNSURE, LIKELY, or CERTAIN (blank if no scoping note) - got '{ p_conf }'.|
      TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
    RETURN.
  ENDIF.

  IF p_topn <= 0.
    MESSAGE 'Candidate Limit must be a positive number.' TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  GET_CONFIG_VALUE
*&---------------------------------------------------------------------*
FORM get_config_value USING iv_key TYPE string iv_default TYPE string
                       CHANGING cv_value TYPE string.
  CLEAR cv_value.
  SELECT SINGLE config_value FROM zdccrt_config INTO cv_value WHERE config_key = iv_key.
  IF sy-subrc <> 0.
    cv_value = iv_default.
  ENDIF.
ENDFORM.

*&=======================================================================*
*& Package validation (hard error on any unknown package)
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  VALIDATE_PACKAGES
*&---------------------------------------------------------------------*
*       Only single-value EQ entries in S_PACK are supported. Each
*       named package is checked individually against ZDCCRT_OBJECTS,
*       so a typo'd package is a hard, immediate error rather than a
*       silently empty candidate set that could otherwise produce a
*       false LIKELY_NEW_OBJECT.
*&---------------------------------------------------------------------*
FORM validate_packages.
  DATA: lt_known TYPE STANDARD TABLE OF devclass,
        lv_found TYPE abap_bool.

  CLEAR: gt_pack_names, gv_error.

  LOOP AT s_pack INTO DATA(ls_pack).
    IF ls_pack-sign <> 'I' OR ls_pack-option <> 'EQ'.
      MESSAGE |Only single package names (no ranges/exclusions) are supported - '{ ls_pack-low }' is not.|
        TYPE 'S' DISPLAY LIKE 'E'.
      gv_error = abap_true.
      RETURN.
    ENDIF.
    APPEND ls_pack-low TO gt_pack_names.
  ENDLOOP.

  SORT gt_pack_names.
  DELETE ADJACENT DUPLICATES FROM gt_pack_names.

  SELECT DISTINCT zpackage FROM zdccrt_objects INTO TABLE lt_known.

  LOOP AT gt_pack_names INTO DATA(lv_pack).
    lv_found = abap_false.
    READ TABLE lt_known TRANSPORTING NO FIELDS WITH KEY table_line = lv_pack.
    IF sy-subrc = 0.
      lv_found = abap_true.
    ENDIF.
    IF lv_found = abap_false.
      MESSAGE |Unknown package '{ lv_pack }' - no rows in ZLLD_OBJECTS. Known packages: { concat_lines_of( table = lt_known sep = ', ' ) }.|
        TYPE 'S' DISPLAY LIKE 'E'.
      gv_error = abap_true.
      RETURN.
    ENDIF.
  ENDLOOP.
ENDFORM.

*&=======================================================================*
*& Object / graph fetching (all package-scoped)
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  FETCH_ALL_OBJECTS
*&---------------------------------------------------------------------*
FORM fetch_all_objects.
  DATA: lt_raw  TYPE STANDARD TABLE OF zdccrt_objects,
        lr_pack TYPE tt_devclass_range.

  CLEAR: gt_objects, gt_object_names.
  PERFORM gt_pack_names_range CHANGING lr_pack.

  SELECT * FROM zdccrt_objects INTO TABLE @lt_raw WHERE zpackage IN @lr_pack.

  LOOP AT lt_raw INTO DATA(ls_raw).
    APPEND VALUE #( name = ls_raw-obj_name type = ls_raw-obj_type package = ls_raw-zpackage ) TO gt_objects.
    APPEND ls_raw-obj_name TO gt_object_names.
  ENDLOOP.

  SORT gt_object_names.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  GT_PACK_NAMES_RANGE
*&---------------------------------------------------------------------*
*       Helper: builds an IN-range table from GT_PACK_NAMES for use in
*       Open SQL WHERE ... IN clauses. Written as a FORM (not a
*       functional method) to stay consistent with this codebase's
*       existing classic-ABAP style - called via the inline functional
*       call syntax where a value, not a CHANGING parameter, is needed.
*&---------------------------------------------------------------------*
FORM gt_pack_names_range CHANGING rt_range TYPE tt_devclass_range.
  CLEAR rt_range.
  LOOP AT gt_pack_names INTO DATA(lv_pack).
    APPEND VALUE #( sign = 'I' option = 'EQ' low = lv_pack ) TO rt_range.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  FETCH_ADJACENCY
*&---------------------------------------------------------------------*
*       Bidirectional adjacency over ZDCCRT_OBJ_CALLS, restricted to
*       edges between two objects BOTH in the given package scope, and
*       to IS_CALL_LIKE edge kinds only. This is joined via
*       ZDCCRT_EKIND_MAP rather than filtering on the stored EDGE_KIND
*       text directly, so a future change to which edge kinds count as
*       call-like needs only a ZDCCRT_EKIND_MAP data change.
*&---------------------------------------------------------------------*
FORM fetch_adjacency.
  DATA: lt_raw TYPE STANDARD TABLE OF ts_edge,
        lr_pack TYPE tt_devclass_range.

  CLEAR gt_adjacency.
  PERFORM gt_pack_names_range CHANGING lr_pack.

  SELECT oc~source_object AS node, oc~target_object AS neighbor
    FROM zdccrt_obj_calls AS oc
    INNER JOIN zdccrt_objects AS o1 ON o1~obj_name = oc~source_object
    INNER JOIN zdccrt_objects AS o2 ON o2~obj_name = oc~target_object
    INNER JOIN zdccrt_ekind_map AS ekm ON ekm~dependency_type = oc~dependency_type
    INTO TABLE @lt_raw
    WHERE ekm~is_call_like = @abap_true
      AND o1~zpackage IN @lr_pack
      AND o2~zpackage IN @lr_pack.

  LOOP AT lt_raw INTO DATA(ls_edge).
    INSERT VALUE #( node = ls_edge-node neighbor = ls_edge-neighbor ) INTO TABLE gt_adjacency.
    INSERT VALUE #( node = ls_edge-neighbor neighbor = ls_edge-node ) INTO TABLE gt_adjacency.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  FETCH_OBJECT_TABLES
*&---------------------------------------------------------------------*
*       Object -> DDIC TABLE names, scoped to DDIC_TYPE = 'TABL' only.
*       The table-sharing bonus is scoped to real database tables;
*       STRU/DTEL are deliberately excluded.
*&---------------------------------------------------------------------*
FORM fetch_object_tables.
  DATA: lt_raw TYPE STANDARD TABLE OF ts_obj_table,
        lr_pack TYPE tt_devclass_range.

  CLEAR gt_obj_tables.
  PERFORM gt_pack_names_range CHANGING lr_pack.

  SELECT dr~obj_name AS object_name, dr~ddic_object_name AS table_name
    FROM zdccrt_obj_ddref AS dr
    INNER JOIN zdccrt_objects AS o ON o~obj_name = dr~obj_name
    INTO TABLE @lt_raw
    WHERE dr~ddic_type = 'TABL'
      AND o~zpackage IN @lr_pack.

  LOOP AT lt_raw INTO DATA(ls_row).
    INSERT ls_row INTO TABLE gt_obj_tables.
  ENDLOOP.
ENDFORM.

*&=======================================================================*
*& Tokenization (must match ZLLD_EXTRACT_TO_DB's tokenizer exactly, or
*& the requirement vector and stored chunk-token vectors live in
*& different vocabularies and cosine similarity is meaningless)
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  BUILD_STOPWORD_LIST
*&---------------------------------------------------------------------*
FORM build_stopword_list.
  DATA: lv_csv TYPE string,
        lt_raw TYPE string_t.

  CLEAR gt_stopwords.
  PERFORM get_config_value USING 'STOPWORDS' '' CHANGING lv_csv.
  SPLIT lv_csv AT ',' INTO TABLE lt_raw.

  LOOP AT lt_raw INTO DATA(lv_word).
    lv_word = to_lower( shift_left( val = lv_word sub = space ) ).
    CONDENSE lv_word.
    IF lv_word IS NOT INITIAL.
      APPEND lv_word TO gt_stopwords.
    ENDIF.
  ENDLOOP.

  SORT gt_stopwords.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  TOKENIZE_TEXT
*&---------------------------------------------------------------------*
*       Same regex/lowercase/stopword-filter/term-frequency logic as
*       ZLLD_EXTRACT_TO_DB's ZLLD_TOKENIZE_CHUNK - deliberately
*       duplicated here rather than shared, since this report and the
*       function module's forms include are separate ABAP program units
*       (same "near-duplicate forms across independently-triggered
*       units" precedent already established between file/DB mode in
*       ZLLD_PACKAGE_EXTRACTOR). Keep both in sync if the tokenization
*       rule ever changes - they MUST agree, or scores become meaningless.
*&---------------------------------------------------------------------*
FORM tokenize_text USING iv_text TYPE string
                    CHANGING ct_tf TYPE tt_token_tf.
  DATA: lt_matches TYPE match_result_tab,
        lv_token   TYPE string.

  CLEAR ct_tf.
  FIND ALL OCCURRENCES OF REGEX '[A-Za-z0-9]+' IN iv_text RESULTS lt_matches.

  LOOP AT lt_matches INTO DATA(ls_match).
    lv_token = to_lower( substring( val = iv_text off = ls_match-offset len = ls_match-length ) ).

    " Purely numeric tokens carry no domain meaning for text similarity -
    " see ZLLD_TOKENIZE_CHUNK's matching check; must stay in lockstep.
    IF lv_token CO '0123456789'.
      CONTINUE.
    ENDIF.

    READ TABLE gt_stopwords TRANSPORTING NO FIELDS WITH KEY table_line = lv_token BINARY SEARCH.
    IF sy-subrc = 0.
      CONTINUE.
    ENDIF.

    READ TABLE ct_tf ASSIGNING FIELD-SYMBOL(<fs>) WITH KEY token = lv_token.
    IF sy-subrc = 0.
      <fs>-tf = <fs>-tf + 1.
    ELSE.
      INSERT VALUE #( token = lv_token tf = 1 ) INTO TABLE ct_tf.
    ENDIF.
  ENDLOOP.
ENDFORM.

*&=======================================================================*
*& TF-IDF cosine similarity
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  COMPUTE_IDF_TABLE
*&---------------------------------------------------------------------*
*       IDF(token) = LOG(total_chunk_count / distinct_chunk_count
*       containing token), computed at query time - but only ever for
*       the requirement's OWN distinct tokens, never the whole scoped
*       package's vocabulary. A token IDF is never looked up for
*       anything outside the requirement's own words (see
*       COMPUTE_REQUIREMENT_VECTOR/COMPUTE_SEMANTIC_SCORES), so scoping
*       the query this way changes nothing about the result - it only
*       stops pulling every chunk/token in the package into ABAP memory
*       to compute a handful of values. Both aggregates (COUNT(*) for
*       the total, GROUP BY token for doc_freq) run server-side, so cost
*       tracks the requirement's vocabulary size, not corpus size.
*       ZDCCRT_CHUNK_TOK's key is already (TOKEN, CHUNK_ID), so each row
*       IS one distinct (token, chunk) pair - COUNT(*) per token already
*       gives doc_freq directly, no DISTINCT needed.
*
*       Chunks shorter than GC_MIN_CHUNK_LENGTH characters are excluded
*       from the scoring corpus entirely (CHUNK_LENGTH, not STRLEN(), so
*       this can run in the WHERE clause) - a second line of defense
*       alongside ZLLD_EXTRACT_TO_DB's own chunk-merge step, not a
*       substitute for it. Must match ZLLD_COMPUTE_CHUNK_NORM's own
*       filter exactly, or a chunk's precomputed CHUNK_NORM would be
*       comparing against a differently-scoped corpus than this IDF is.
*&---------------------------------------------------------------------*
FORM compute_idf_table.
  DATA: lt_req_tf TYPE tt_token_tf,
        lr_token  TYPE RANGE OF string,
        lr_pack   TYPE tt_devclass_range.

  CLEAR: gt_idf, gv_total_chunks.
  PERFORM gt_pack_names_range CHANGING lr_pack.

  " Tokenized here (and again, separately, in COMPUTE_REQUIREMENT_VECTOR)
  " purely to know which tokens to scope this query to - the requirement
  " text is a few sentences at most, so retokenizing it twice costs
  " nothing next to what it saves on the corpus side.
  PERFORM tokenize_text USING gv_requirement_text CHANGING lt_req_tf.
  CHECK lt_req_tf IS NOT INITIAL.

  LOOP AT lt_req_tf INTO DATA(ls_tf).
    APPEND VALUE #( sign = 'I' option = 'EQ' low = ls_tf-token ) TO lr_token.
  ENDLOOP.

  SELECT COUNT(*)
    FROM zdccrt_chunks AS c
    INNER JOIN zdccrt_objects AS o ON o~obj_name = c~obj_name
    WHERE o~zpackage IN @lr_pack
      AND c~chunk_length >= @gv_min_chunk_length
    INTO @gv_total_chunks.

  CHECK gv_total_chunks > 0.

  SELECT tok~token AS token, COUNT( * ) AS doc_freq
    FROM zdccrt_chunk_tok AS tok
    INNER JOIN zdccrt_chunks  AS c ON c~chunk_id = tok~chunk_id
    INNER JOIN zdccrt_objects AS o ON o~obj_name = c~obj_name
    INTO TABLE @DATA(lt_raw_freq)
    WHERE tok~token IN @lr_token
      AND o~zpackage IN @lr_pack
      AND c~chunk_length >= @gv_min_chunk_length
    GROUP BY tok~token.

  LOOP AT lt_raw_freq INTO DATA(ls_df).
    CHECK ls_df-doc_freq > 0.
    INSERT VALUE #( token = ls_df-token
                     idf   = log( gv_total_chunks / ls_df-doc_freq ) )
      INTO TABLE gt_idf.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  COMPUTE_REQUIREMENT_VECTOR
*&---------------------------------------------------------------------*
*       Requirement tokens with no IDF at all (never appear in any
*       scoped chunk) get weight 0 and are excluded from the requirement
*       vector entirely - such a token can never match anything anyway,
*       and including it with an undefined/zero IDF would only inflate
*       the requirement's own norm for no reason. GV_REQ_NORM is the
*       requirement vector's Euclidean norm, computed once and reused
*       for every chunk's cosine denominator.
*&---------------------------------------------------------------------*
FORM compute_requirement_vector.
  DATA: lt_req_tf  TYPE tt_token_tf,
        lv_idf     TYPE f,
        lv_norm_sq TYPE f.

  CLEAR: gt_req_weighted, gv_req_norm.
  PERFORM tokenize_text USING gv_requirement_text CHANGING lt_req_tf.

  LOOP AT lt_req_tf INTO DATA(ls_tf).
    READ TABLE gt_idf INTO DATA(ls_idf) WITH KEY token = ls_tf-token BINARY SEARCH.
    IF sy-subrc = 0.
      lv_idf = ls_idf-idf.
    ELSE.
      CONTINUE.
    ENDIF.

    DATA(lv_weight) = ls_tf-tf * lv_idf.
    INSERT VALUE #( token = ls_tf-token weight = lv_weight ) INTO TABLE gt_req_weighted.
    lv_norm_sq = lv_norm_sq + lv_weight * lv_weight.
  ENDLOOP.

  gv_req_norm = sqrt( lv_norm_sq ).
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  FINALIZE_CHUNK_MATCH
*&---------------------------------------------------------------------*
*       Turns one chunk's accumulated dot product into a cosine score
*       and, if it beats that object's current best, updates CT_BEST.
*       Unlike the old norm_sq-accumulation approach, the chunk's own
*       norm is a lookup (IT_CHUNK_NORMS, precomputed at extraction time
*       - see ZLLD_COMPUTE_CHUNK_NORM), never derived here - this pass
*       only ever sees the chunk's requirement-overlapping tokens, not
*       its complete token set, so it has no way to compute a norm on
*       its own even if it wanted to.
*&---------------------------------------------------------------------*
FORM finalize_chunk_match USING iv_object      TYPE string
                                 iv_dot         TYPE f
                                 iv_chunk_id    TYPE sysuuid_x16
                                 it_chunk_norms TYPE tt_chunk_norm
                           CHANGING ct_best     TYPE tt_score.
  DATA lv_cosine TYPE f.

  CHECK iv_object IS NOT INITIAL.

  READ TABLE it_chunk_norms INTO DATA(ls_norm) WITH TABLE KEY chunk_id = iv_chunk_id.
  DATA(lv_chunk_norm) = COND #( WHEN sy-subrc = 0 THEN ls_norm-chunk_norm ELSE 0 ).

  lv_cosine = COND #( WHEN gv_req_norm > 0 AND lv_chunk_norm > 0
                       THEN iv_dot / ( gv_req_norm * lv_chunk_norm )
                       ELSE 0 ).

  READ TABLE ct_best ASSIGNING FIELD-SYMBOL(<fs_best>) WITH KEY object_name = iv_object.
  IF sy-subrc = 0.
    IF lv_cosine > <fs_best>-score.
      <fs_best>-score = lv_cosine.
    ENDIF.
  ELSE.
    INSERT VALUE #( object_name = iv_object score = lv_cosine ) INTO TABLE ct_best.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  COMPUTE_SEMANTIC_SCORES
*&---------------------------------------------------------------------*
*       Max cosine similarity across an object's chunks - one strong
*       matching chunk in a large object is real signal, so the maximum
*       is used rather than an average.
*
*       Only fetches ZDCCRT_CHUNK_TOK rows for chunks that share at
*       least one word with the requirement (WHERE TOKEN IN GT_REQ_
*       WEIGHTED's own tokens) - every other chunk's cosine is 0 by
*       definition (a zero requirement-weight token contributes 0 to the
*       dot product regardless of the chunk's own weight for it), so
*       there is nothing to gain by loading them. The chunk's own norm
*       (which DOES depend on its complete token set) is a lookup
*       against ZDCCRT_CHUNKS-CHUNK_NORM instead, precomputed once at
*       extraction time - see ZLLD_COMPUTE_CHUNK_NORM.
*
*       Group boundaries are detected with an explicit "did CHUNK_ID
*       change since the last row" comparison, not AT NEW/AT END OF
*       CHUNK_ID - see this codebase's own history with that control-
*       break rule firing on the wrong field (ABAP's "AT NEW f" also
*       fires on any field to f's LEFT changing).
*&---------------------------------------------------------------------*
FORM compute_semantic_scores.
  DATA: lr_token        TYPE RANGE OF string,
        lr_pack         TYPE tt_devclass_range,
        lt_matches      TYPE tt_semantic_match,
        lt_chunk_ids    TYPE STANDARD TABLE OF sysuuid_x16,
        lt_chunk_norms  TYPE tt_chunk_norm,
        lv_dot          TYPE f,
        lv_idf          TYPE f,
        lv_weight       TYPE f,
        lv_cur_chunk_id TYPE sysuuid_x16,
        lv_cur_object   TYPE string,
        lv_group_open   TYPE abap_bool,
        lt_best         TYPE tt_score.

  CLEAR gt_obj_score.

  IF gt_req_weighted IS NOT INITIAL.
    LOOP AT gt_req_weighted INTO DATA(ls_rw).
      APPEND VALUE #( sign = 'I' option = 'EQ' low = ls_rw-token ) TO lr_token.
    ENDLOOP.
    PERFORM gt_pack_names_range CHANGING lr_pack.

    SELECT tok~token AS token, tok~chunk_id AS chunk_id, tok~term_frequency AS tf,
           c~obj_name AS object_name
      FROM zdccrt_chunk_tok AS tok
      INNER JOIN zdccrt_chunks  AS c ON c~chunk_id = tok~chunk_id
      INNER JOIN zdccrt_objects AS o ON o~obj_name = c~obj_name
      INTO TABLE @lt_matches
      WHERE tok~token IN @lr_token
        AND o~zpackage IN @lr_pack
        AND c~chunk_length >= @gv_min_chunk_length.

    IF lt_matches IS NOT INITIAL.
      LOOP AT lt_matches INTO DATA(ls_id_scan).
        APPEND ls_id_scan-chunk_id TO lt_chunk_ids.
      ENDLOOP.
      SORT lt_chunk_ids.
      DELETE ADJACENT DUPLICATES FROM lt_chunk_ids.

      SELECT chunk_id, chunk_norm FROM zdccrt_chunks
        INTO TABLE @lt_chunk_norms
        FOR ALL ENTRIES IN @lt_chunk_ids
        WHERE chunk_id = @lt_chunk_ids-table_line.
    ENDIF.

    SORT lt_matches BY chunk_id.

    LOOP AT lt_matches INTO DATA(ls_row).
      IF lv_group_open = abap_false OR ls_row-chunk_id <> lv_cur_chunk_id.
        IF lv_group_open = abap_true.
          PERFORM finalize_chunk_match USING lv_cur_object lv_dot lv_cur_chunk_id lt_chunk_norms
                                        CHANGING lt_best.
        ENDIF.

        lv_dot = 0.
        lv_cur_chunk_id = ls_row-chunk_id.
        lv_cur_object   = ls_row-object_name.
        lv_group_open = abap_true.
      ENDIF.

      READ TABLE gt_idf INTO DATA(ls_idf) WITH KEY token = ls_row-token BINARY SEARCH.
      lv_idf = COND #( WHEN sy-subrc = 0 THEN ls_idf-idf ELSE 0 ).
      lv_weight = ls_row-tf * lv_idf.

      READ TABLE gt_req_weighted INTO DATA(ls_req) WITH KEY token = ls_row-token BINARY SEARCH.
      IF sy-subrc = 0.
        lv_dot = lv_dot + lv_weight * ls_req-weight.
      ENDIF.
    ENDLOOP.

    " Finalize the last group in the table (no further row triggers it).
    IF lv_group_open = abap_true.
      PERFORM finalize_chunk_match USING lv_cur_object lv_dot lv_cur_chunk_id lt_chunk_norms
                                    CHANGING lt_best.
    ENDIF.
  ENDIF.

  " Seed GT_OBJ_SCORE with every object in scope (score 0 for objects
  " with no matching chunk at all), then overlay the best cosine found.
  LOOP AT gt_objects INTO DATA(ls_obj).
    INSERT VALUE #( object_name = ls_obj-name type = ls_obj-type package = ls_obj-package semantic = 0 )
      INTO TABLE gt_obj_score.
  ENDLOOP.

  LOOP AT lt_best INTO DATA(ls_best).
    READ TABLE gt_obj_score ASSIGNING FIELD-SYMBOL(<fs_os>) WITH KEY object_name = ls_best-object_name.
    IF sy-subrc = 0.
      <fs_os>-semantic = ls_best-score.
    ENDIF.
  ENDLOOP.
ENDFORM.

*&=======================================================================*
*& Structural scoring (BFS + table-sharing bonus)
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  BFS_HOP_DISTANCES
*&---------------------------------------------------------------------*
*       Multi-source BFS over GT_ADJACENCY. Returns {node: hop_distance}
*       for every node reached within IV_MAX_DEPTH hops of any seed,
*       seeds themselves at 0 - direct port of _bfs_hop_distances.
*&---------------------------------------------------------------------*
FORM bfs_hop_distances USING it_seeds TYPE string_t iv_max_depth TYPE i
                        CHANGING ct_dist TYPE tt_node_dist.
  DATA: lt_frontier      TYPE string_t,
        lt_next_frontier TYPE string_t,
        lv_depth         TYPE i.

  CLEAR ct_dist.

  LOOP AT it_seeds INTO DATA(lv_seed).
    INSERT VALUE #( node = lv_seed dist = 0 ) INTO TABLE ct_dist.
  ENDLOOP.
  lt_frontier = it_seeds.

  DO iv_max_depth TIMES.
    lv_depth = sy-index.
    CLEAR lt_next_frontier.

    LOOP AT lt_frontier INTO DATA(lv_node).
      LOOP AT gt_adjacency INTO DATA(ls_edge) WHERE node = lv_node.
        READ TABLE ct_dist TRANSPORTING NO FIELDS WITH KEY node = ls_edge-neighbor.
        IF sy-subrc <> 0.
          INSERT VALUE #( node = ls_edge-neighbor dist = lv_depth ) INTO TABLE ct_dist.
          APPEND ls_edge-neighbor TO lt_next_frontier.
        ENDIF.
      ENDLOOP.
    ENDLOOP.

    lt_frontier = lt_next_frontier.
    IF lt_frontier IS INITIAL.
      EXIT.
    ENDIF.
  ENDDO.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  COMPUTE_SEED_STRUCT_SCORES
*&---------------------------------------------------------------------*
*       Top-3 by semantic score, gated by GV_SEED_MIN_SEMANTIC: an
*       object only becomes a structural seed if its own semantic score
*       clears this threshold. Without the gate, the top-3 objects would
*       always get structural_score=1.0 via self-membership regardless
*       of how weak their real semantic match is, making
*       likely_new_object mathematically unreachable. Structural score
*       is max(hop-decay, table-sharing bonus), never summed.
*
*       GT_NAMED (the developer's resolved scoping note, if any) is
*       UNIONED into the seed list, ungated by GV_SEED_MIN_SEMANTIC. A
*       developer naming an object directly is stronger evidence than a
*       coincidental top-3 finish, not weaker, so it does not need to
*       additionally clear the same gate designed to filter out weak
*       automatic candidates. This requires RESOLVE_NOTE_AND_NEIGHBORS
*       to run before this FORM (see MAIN) so GT_NAMED is already
*       populated here.
*&---------------------------------------------------------------------*
FORM compute_seed_struct_scores.
  DATA: lt_sorted_by_semantic TYPE STANDARD TABLE OF ts_obj_score,
        lt_top3               TYPE string_t,
        lt_seeds              TYPE string_t,
        lt_seed_tables        TYPE string_t,
        lt_dist               TYPE tt_node_dist,
        lv_hop_score          TYPE f,
        lv_table_score        TYPE f,
        lv_shares_table       TYPE abap_bool.

  lt_sorted_by_semantic = gt_obj_score.
  SORT lt_sorted_by_semantic BY semantic DESCENDING object_name ASCENDING.

  LOOP AT lt_sorted_by_semantic INTO DATA(ls_top) TO gv_top_seed_objects.
    APPEND ls_top-object_name TO lt_top3.
  ENDLOOP.

  LOOP AT lt_top3 INTO DATA(lv_cand).
    READ TABLE gt_obj_score INTO DATA(ls_score) WITH KEY object_name = lv_cand.
    IF sy-subrc = 0 AND ls_score-semantic >= gv_seed_min_semantic.
      APPEND lv_cand TO lt_seeds.
    ENDIF.
  ENDLOOP.

  APPEND LINES OF gt_named TO lt_seeds.
  SORT lt_seeds. DELETE ADJACENT DUPLICATES FROM lt_seeds.

  PERFORM bfs_hop_distances USING lt_seeds 2 CHANGING lt_dist.

  " Seed tables: union of every seed's own DDIC TABLE references.
  LOOP AT lt_seeds INTO DATA(lv_seed_name).
    LOOP AT gt_obj_tables INTO DATA(ls_st) WHERE object_name = lv_seed_name.
      APPEND ls_st-table_name TO lt_seed_tables.
    ENDLOOP.
  ENDLOOP.
  SORT lt_seed_tables.
  DELETE ADJACENT DUPLICATES FROM lt_seed_tables.

  LOOP AT gt_obj_score ASSIGNING FIELD-SYMBOL(<fs_os>).
    READ TABLE lt_seeds TRANSPORTING NO FIELDS WITH KEY table_line = <fs_os>-object_name.
    IF sy-subrc = 0.
      lv_hop_score = gv_structural_seed_score.
    ELSE.
      READ TABLE lt_dist INTO DATA(ls_dist) WITH KEY node = <fs_os>-object_name.
      IF sy-subrc = 0.
        CASE ls_dist-dist.
          WHEN 1. lv_hop_score = gv_structural_hop1_score.
          WHEN 2. lv_hop_score = gv_structural_hop2_score.
          WHEN OTHERS. lv_hop_score = 0.
        ENDCASE.
      ELSE.
        lv_hop_score = 0.
      ENDIF.
    ENDIF.

    lv_shares_table = abap_false.
    LOOP AT gt_obj_tables INTO DATA(ls_ot) WHERE object_name = <fs_os>-object_name.
      READ TABLE lt_seed_tables TRANSPORTING NO FIELDS WITH KEY table_line = ls_ot-table_name.
      IF sy-subrc = 0.
        lv_shares_table = abap_true.
        EXIT.
      ENDIF.
    ENDLOOP.
    lv_table_score = COND #( WHEN lv_shares_table = abap_true THEN gv_structural_shared_tbl ELSE 0 ).

    <fs_os>-structural = nmax( val1 = lv_hop_score val2 = lv_table_score ).
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  COMPUTE_BASE_SCORES
*&---------------------------------------------------------------------*
FORM compute_base_scores.
  LOOP AT gt_obj_score ASSIGNING FIELD-SYMBOL(<fs_os>).
    <fs_os>-base_score = gv_semantic_weight * <fs_os>-semantic + gv_structural_weight * <fs_os>-structural.
  ENDLOOP.
ENDFORM.

*&=======================================================================*
*& Scoping note resolution, tier adjustment, ranking
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  RESOLVE_NOTE_AND_NEIGHBORS
*&---------------------------------------------------------------------*
*       Exact match first, then case-insensitive, then substring (either
*       direction). An unresolved note is recorded as a disagreement,
*       not an error - useful signal for the developer, not a failure.
*       GV_EFFECTIVE_CONFIDENCE stays blank in that case (nothing to
*       anchor a tier adjustment to).
*&---------------------------------------------------------------------*
FORM resolve_note_and_neighbors.
  DATA: lt_exact TYPE string_t,
        lt_ci    TYPE string_t,
        lt_sub   TYPE string_t,
        lv_note_lower TYPE string,
        lt_dist  TYPE tt_node_dist.

  CLEAR: gt_named, gt_named_hop1, gt_named_certain, gv_effective_confidence.

  IF p_note IS NOT INITIAL.
    LOOP AT gt_object_names INTO DATA(lv_name).
      IF lv_name = p_note.
        APPEND lv_name TO lt_exact.
      ENDIF.
    ENDLOOP.

    IF lt_exact IS INITIAL.
      lv_note_lower = to_lower( p_note ).
      LOOP AT gt_object_names INTO lv_name.
        IF to_lower( lv_name ) = lv_note_lower.
          APPEND lv_name TO lt_ci.
        ENDIF.
      ENDLOOP.
    ENDIF.

    IF lt_exact IS INITIAL AND lt_ci IS INITIAL.
      lv_note_lower = to_lower( p_note ).
      LOOP AT gt_object_names INTO lv_name.
        IF to_lower( lv_name ) CS lv_note_lower OR lv_note_lower CS to_lower( lv_name ).
          APPEND lv_name TO lt_sub.
        ENDIF.
      ENDLOOP.
    ENDIF.

    IF lt_exact IS NOT INITIAL.
      gt_named = lt_exact.
    ELSEIF lt_ci IS NOT INITIAL.
      gt_named = lt_ci.
    ELSE.
      gt_named = lt_sub.
    ENDIF.
    SORT gt_named. DELETE ADJACENT DUPLICATES FROM gt_named.
  ENDIF.

  IF p_note IS NOT INITIAL AND gt_named IS INITIAL.
    " INDEX 1 (not INTO TABLE - that combination is invalid; INDEX is
    " the positional form, INTO TABLE the key-based form, never both)
    " so this unresolvable-note disagreement sorts first, ahead of any
    " per-entry disagreements appended later in the pipeline.
    INSERT VALUE #(
      kind = 'unresolvable_scoping_note'
      message = |Scoping note '{ p_note }' did not match any known object (checked exact, | &&
                |case-insensitive, and substring match). This may mean the developer has an | &&
                |object in mind that is outside the extracted package(s).| )
      INTO gt_disagreements INDEX 1.
  ENDIF.

  gv_effective_confidence = COND #( WHEN gt_named IS NOT INITIAL THEN p_conf ELSE '' ).

  CHECK gt_named IS NOT INITIAL.

  " Exactly 1 hop (direct adjacency only), used by the LIKELY tier's
  " hop-1 multiplier - independent of the separately-configurable
  " certain-filter radius below (both happen to be 1 by default).
  LOOP AT gt_named INTO DATA(lv_named).
    LOOP AT gt_adjacency INTO DATA(ls_edge) WHERE node = lv_named.
      APPEND ls_edge-neighbor TO gt_named_hop1.
    ENDLOOP.
  ENDLOOP.
  SORT gt_named_hop1. DELETE ADJACENT DUPLICATES FROM gt_named_hop1.
  LOOP AT gt_named INTO lv_named.
    DELETE gt_named_hop1 WHERE table_line = lv_named.
  ENDLOOP.

  PERFORM bfs_hop_distances USING gt_named gv_certain_hop_radius CHANGING lt_dist.
  LOOP AT lt_dist INTO DATA(ls_d).
    APPEND ls_d-node TO gt_named_certain.
  ENDLOOP.
  LOOP AT gt_named INTO lv_named.
    DELETE gt_named_certain WHERE table_line = lv_named.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_UNADJUSTED_RANKING
*&---------------------------------------------------------------------*
*       Pure BASE_SCORE ranking, always over every object in scope -
*       used both as the top-5 "unadjusted_ranking" output and as the
*       comparison baseline for disagreement detection.
*&---------------------------------------------------------------------*
FORM build_unadjusted_ranking.
  DATA lt_sorted TYPE tt_obj_score_std.

  CLEAR gt_unadjusted_sorted.
  lt_sorted = gt_obj_score.
  SORT lt_sorted BY base_score DESCENDING object_name ASCENDING.

  LOOP AT lt_sorted INTO DATA(ls_row).
    APPEND ls_row-object_name TO gt_unadjusted_sorted.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  APPLY_TIER_ADJUSTMENT
*&---------------------------------------------------------------------*
*       Direct port of retrieve()'s tier-adjustment block: CERTAIN is a
*       hard filter (no multiplier, GT_ALLOWED restricts the candidate
*       universe); LIKELY/UNSURE apply multipliers without filtering;
*       no tier leaves FINAL_SCORE = BASE_SCORE for everyone.
*&---------------------------------------------------------------------*
FORM apply_tier_adjustment.
  CLEAR: gt_allowed, gv_has_filter.

  CASE gv_effective_confidence.
    WHEN 'CERTAIN'.
      gv_has_filter = abap_true.
      gt_allowed = gt_named.
      APPEND LINES OF gt_named_certain TO gt_allowed.
      SORT gt_allowed. DELETE ADJACENT DUPLICATES FROM gt_allowed.

      LOOP AT gt_obj_score ASSIGNING FIELD-SYMBOL(<fs>).
        <fs>-final_score = <fs>-base_score.
        READ TABLE gt_named TRANSPORTING NO FIELDS WITH KEY table_line = <fs>-object_name.
        IF sy-subrc = 0.
          <fs>-tier_label = 'certain: in filter (named object)'.
        ELSE.
          READ TABLE gt_named_certain TRANSPORTING NO FIELDS WITH KEY table_line = <fs>-object_name.
          IF sy-subrc = 0.
            <fs>-tier_label = |certain: in filter (within { gv_certain_hop_radius } hop(s) of named)|.
          ELSE.
            <fs>-tier_label = 'certain: excluded (outside filter)'.
          ENDIF.
        ENDIF.
      ENDLOOP.

    WHEN 'LIKELY'.
      LOOP AT gt_obj_score ASSIGNING FIELD-SYMBOL(<fs2>).
        READ TABLE gt_named TRANSPORTING NO FIELDS WITH KEY table_line = <fs2>-object_name.
        IF sy-subrc = 0.
          <fs2>-final_score = <fs2>-base_score * gv_likely_named_mult.
          <fs2>-tier_label  = |likely: x{ gv_likely_named_mult } (named object)|.
        ELSE.
          READ TABLE gt_named_hop1 TRANSPORTING NO FIELDS WITH KEY table_line = <fs2>-object_name.
          IF sy-subrc = 0.
            <fs2>-final_score = <fs2>-base_score * gv_likely_hop1_mult.
            <fs2>-tier_label  = |likely: x{ gv_likely_hop1_mult } (1-hop of named)|.
          ELSE.
            <fs2>-final_score = <fs2>-base_score.
            <fs2>-tier_label  = 'likely: no boost'.
          ENDIF.
        ENDIF.
      ENDLOOP.

    WHEN 'UNSURE'.
      LOOP AT gt_obj_score ASSIGNING FIELD-SYMBOL(<fs3>).
        READ TABLE gt_named TRANSPORTING NO FIELDS WITH KEY table_line = <fs3>-object_name.
        IF sy-subrc = 0.
          <fs3>-final_score = <fs3>-base_score * gv_unsure_named_mult.
          <fs3>-tier_label  = |unsure: x{ gv_unsure_named_mult } (named object)|.
        ELSE.
          <fs3>-final_score = <fs3>-base_score.
          <fs3>-tier_label  = 'unsure: no boost'.
        ENDIF.
      ENDLOOP.

    WHEN OTHERS.
      LOOP AT gt_obj_score ASSIGNING FIELD-SYMBOL(<fs4>).
        <fs4>-final_score = <fs4>-base_score.
        <fs4>-tier_label  = 'none'.
      ENDLOOP.
  ENDCASE.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_ADJUSTED_RANKING
*&---------------------------------------------------------------------*
*       GT_ADJUSTED_SORTED is the FULL sorted candidate universe (all
*       objects, or just GT_ALLOWED under CERTAIN) - used both for the
*       exported top-N candidates and, unsliced, for new-object
*       detection's top/5th-candidate spread check.
*&---------------------------------------------------------------------*
FORM build_adjusted_ranking.
  DATA: lt_universe TYPE tt_obj_score,
        lt_sorted   TYPE tt_obj_score_std.

  CLEAR gt_adjusted_sorted.

  IF gv_has_filter = abap_true.
    LOOP AT gt_obj_score INTO DATA(ls_row).
      READ TABLE gt_allowed TRANSPORTING NO FIELDS WITH KEY table_line = ls_row-object_name.
      IF sy-subrc = 0.
        INSERT ls_row INTO TABLE lt_universe.
      ENDIF.
    ENDLOOP.
  ELSE.
    lt_universe = gt_obj_score.
  ENDIF.

  lt_sorted = lt_universe.
  SORT lt_sorted BY final_score DESCENDING object_name ASCENDING.

  LOOP AT lt_sorted INTO DATA(ls_sorted).
    APPEND ls_sorted-object_name TO gt_adjusted_sorted.
  ENDLOOP.
ENDFORM.

*&=======================================================================*
*& Disagreement detection (direct port, both rules)
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  DETECT_CERTAIN_HINT_CONFLICT
*&---------------------------------------------------------------------*
FORM detect_certain_hint_conflict.
  DATA: lv_best_in_filter TYPE f,
        lv_top_unadjusted TYPE string,
        lv_gap            TYPE f.

  CHECK gv_effective_confidence = 'CERTAIN' AND gv_has_filter = abap_true.

  LOOP AT gt_obj_score INTO DATA(ls_row).
    READ TABLE gt_allowed TRANSPORTING NO FIELDS WITH KEY table_line = ls_row-object_name.
    IF sy-subrc = 0 AND ls_row-base_score > lv_best_in_filter.
      lv_best_in_filter = ls_row-base_score.
    ENDIF.
  ENDLOOP.

  READ TABLE gt_unadjusted_sorted INTO lv_top_unadjusted INDEX 1.
  CHECK sy-subrc = 0.

  READ TABLE gt_allowed TRANSPORTING NO FIELDS WITH KEY table_line = lv_top_unadjusted.
  CHECK sy-subrc <> 0.  " top unadjusted object is NOT in the filter

  READ TABLE gt_obj_score INTO DATA(ls_top) WITH KEY object_name = lv_top_unadjusted.
  CHECK sy-subrc = 0.

  lv_gap = ls_top-base_score - lv_best_in_filter.
  CHECK lv_gap > gv_certain_conflict_margin.

  INSERT VALUE #(
    kind = 'certain_hint_conflict'
    message = |Certain scoping note restricts results to { concat_lines_of( table = gt_allowed sep = ', ' ) }, | &&
              |but '{ lv_top_unadjusted }' (base_score={ ls_top-base_score DECIMALS = 3 }) scores | &&
              |{ lv_gap DECIMALS = 3 } higher than the best in-filter object | &&
              |(base_score={ lv_best_in_filter DECIMALS = 3 }). The hard filter may be hiding the real answer.| )
    INTO TABLE gt_disagreements.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  DETECT_TIER_OVERRIDE
*&---------------------------------------------------------------------*
FORM detect_tier_override.
  DATA: lv_top_adjusted   TYPE string,
        lv_top_unadjusted TYPE string.

  CHECK gv_effective_confidence = 'LIKELY' OR gv_effective_confidence = 'UNSURE'.

  READ TABLE gt_adjusted_sorted INTO lv_top_adjusted INDEX 1.
  READ TABLE gt_unadjusted_sorted INTO lv_top_unadjusted INDEX 1.
  CHECK lv_top_adjusted <> lv_top_unadjusted.

  READ TABLE gt_obj_score INTO DATA(ls_adj) WITH KEY object_name = lv_top_adjusted.
  READ TABLE gt_obj_score INTO DATA(ls_unadj) WITH KEY object_name = lv_top_unadjusted.

  INSERT VALUE #(
    kind = |{ to_lower( gv_effective_confidence ) }_override|
    message = |With the { to_lower( gv_effective_confidence ) } scoping hint applied, the top result is | &&
              |'{ lv_top_adjusted }' (final_score={ ls_adj-final_score DECIMALS = 3 }), but the top | &&
              |independent match without the hint is '{ lv_top_unadjusted }' | &&
              |(base_score={ ls_unadj-base_score DECIMALS = 3 }). The hint changed the top rank.| )
    INTO TABLE gt_disagreements.
ENDFORM.

*&=======================================================================*
*& New-object detection (uses BASE_SCORE, not the tier-adjusted
*& FINAL_SCORE, so a scoping-note boost never masks a genuinely weak
*& match)
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  DETECT_NEW_OBJECT
*&---------------------------------------------------------------------*
FORM detect_new_object.
  DATA: lv_count      TYPE i,
        lv_top_base   TYPE f,
        lv_fifth_idx  TYPE i,
        lv_fifth_base TYPE f,
        lv_spread     TYPE f,
        lv_pack_list  TYPE string.

  CLEAR: gv_likely_new_object, gv_new_object_reason.
  lv_pack_list = concat_lines_of( table = gt_pack_names sep = ', ' ).

  lv_count = lines( gt_adjusted_sorted ).
  IF lv_count = 0.
    gv_likely_new_object = abap_true.
    gv_new_object_reason = |No candidates available at all in package(s) { lv_pack_list }.|.
    RETURN.
  ENDIF.

  READ TABLE gt_adjusted_sorted INTO DATA(lv_top_name) INDEX 1.
  READ TABLE gt_obj_score INTO DATA(ls_top) WITH KEY object_name = lv_top_name.
  lv_top_base = ls_top-base_score.

  lv_fifth_idx = nmin( val1 = 5 val2 = lv_count ).
  READ TABLE gt_adjusted_sorted INTO DATA(lv_fifth_name) INDEX lv_fifth_idx.
  READ TABLE gt_obj_score INTO DATA(ls_fifth) WITH KEY object_name = lv_fifth_name.
  lv_fifth_base = ls_fifth-base_score.

  lv_spread = lv_top_base - lv_fifth_base.

  IF lv_top_base < gv_new_object_floor.
    gv_likely_new_object = abap_true.
    gv_new_object_reason = |Top candidate's base_score ({ lv_top_base DECIMALS = 3 }) is below the floor | &&
                            |({ gv_new_object_floor }) in package(s) { lv_pack_list }. This may be new work, | &&
                            |or it may belong to a package outside the given scope.|.
  ELSEIF lv_count >= 2 AND lv_spread < gv_new_object_spread.
    gv_likely_new_object = abap_true.
    gv_new_object_reason = |Spread between top and #{ lv_fifth_idx } candidate base_score | &&
                            |({ lv_spread DECIMALS = 3 }) is below threshold ({ gv_new_object_spread }) in | &&
                            |package(s) { lv_pack_list } - nothing stands out. This may be new work, or it | &&
                            |may belong to a package outside the given scope.|.
  ENDIF.
ENDFORM.

*&=======================================================================*
*& Export - writes the candidate file in the format the downstream
*& gem tool reads
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  JSON_ESCAPE_STRING
*&---------------------------------------------------------------------*
*       IV_RAW is TYPE CSEQUENCE (a generic ABAP type covering STRING
*       and any fixed CHAR length), not a fixed TYPE STRING: callers
*       pass a mix of genuinely STRING actuals and CHAR-length actuals
*       (DEPENDENCY_TYPE is CHAR10, TARGET_OBJECT is CHAR60 in
*       ZDCCRT_OBJ_CALLS; OBJECT_TYPE is CHAR4). A classic FORM's default
*       by-reference USING binding requires the actual parameter's type
*       to be IDENTICAL to the formal's, so no single fixed type can
*       match every caller here; CSEQUENCE lets ABAP bind BY REFERENCE
*       directly to whichever concrete text type the caller actually
*       passes, with no value-copy needed.
*&---------------------------------------------------------------------*
FORM json_escape_string USING iv_raw TYPE csequence CHANGING cv_escaped TYPE string.
  cv_escaped = iv_raw.
  REPLACE ALL OCCURRENCES OF '\' IN cv_escaped WITH '\\'.
  REPLACE ALL OCCURRENCES OF '"' IN cv_escaped WITH '\"'.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_DEPENDENCIES_JSON
*&---------------------------------------------------------------------*
*       Reconstructs the real, flat {"TYPE","NAME","SIGNATURE"} shape
*       from ZDCCRT_OBJ_CALLS, using the raw DEPENDENCY_TYPE rather than
*       EDGE_KIND, since the gem's evidence rule is written against the
*       real ABAP type vocabulary. SIGNATURE_JSON is embedded VERBATIM,
*       unescaped - it is already a raw JSON object/array substring
*       copied straight through by ZLLD_EXTRACT_TO_DB, so re-escaping it
*       as a JSON string value would double-encode it and produce a
*       shape the parser/gem does not expect. Outer TYPE uses
*       ZDCCRT_OBJECTS' own short OBJ_TYPE code directly.
*
*       ZDCCRT_OBJ_DDREF is NOT separately unioned in here: every
*       DDIC-shaped dependency ZLLD_EXTRACT_TO_DB staged into
*       ZDCCRT_OBJ_DDREF was ALSO written to ZDCCRT_OBJ_CALLS (see that
*       include's ZLLD_LOAD_ONE_OBJECT form) - reading ZDCCRT_OBJ_CALLS
*       alone already includes every DDIC reference.
*&---------------------------------------------------------------------*
FORM build_dependencies_json USING iv_object_name TYPE string iv_object_type TYPE char4
                              CHANGING cv_json TYPE string.
  DATA: lt_calls    TYPE STANDARD TABLE OF zdccrt_obj_calls,
        lt_entries  TYPE string_t,
        lv_type_esc TYPE string,
        lv_name_esc TYPE string,
        lv_entry    TYPE string,
        lv_outer_type_esc TYPE string,
        lv_outer_name_esc TYPE string.

  SELECT * FROM zdccrt_obj_calls INTO TABLE lt_calls
    WHERE source_object = iv_object_name
    ORDER BY target_object.

  LOOP AT lt_calls INTO DATA(ls_call).
    PERFORM json_escape_string USING ls_call-dependency_type CHANGING lv_type_esc.
    PERFORM json_escape_string USING ls_call-target_object CHANGING lv_name_esc.

    lv_entry = |\{"TYPE":"{ lv_type_esc }","NAME":"{ lv_name_esc }"|.
    IF ls_call-signature_json IS NOT INITIAL.
      lv_entry = |{ lv_entry },"SIGNATURE":{ ls_call-signature_json }|.
    ENDIF.
    lv_entry = |{ lv_entry }\}|.
    APPEND lv_entry TO lt_entries.
  ENDLOOP.

  PERFORM json_escape_string USING iv_object_type CHANGING lv_outer_type_esc.
  PERFORM json_escape_string USING iv_object_name CHANGING lv_outer_name_esc.

  cv_json = |[\{"TYPE":"{ lv_outer_type_esc }","NAME":"{ lv_outer_name_esc }","DEPENDENCIES":[| &&
            |{ concat_lines_of( table = lt_entries sep = `,` ) }]\}]|.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_SOURCE_TEXT
*&---------------------------------------------------------------------*
*       Full source reconstruction by concatenating ZDCCRT_CHUNKS rows
*       in CHUNK_INDEX order - NOT just the matching chunk. The gem
*       needs the entire method body to judge whether a requirement is
*       already implemented, not only the top-scoring fragment. Joined
*       with a blank line between chunks.
*&---------------------------------------------------------------------*
FORM build_source_text USING iv_object_name TYPE string CHANGING cv_source TYPE string.
  DATA lt_texts TYPE string_t.

  SELECT chunk_text FROM zdccrt_chunks INTO TABLE lt_texts
    WHERE obj_name = iv_object_name
    ORDER BY chunk_index.

  cv_source = concat_lines_of( table = lt_texts sep = |\n\n| ).
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_DDIC_JSON_SECTION
*&---------------------------------------------------------------------*
*       Rebuilds the {"PACKAGES":[{"PACKAGE":...,"DOMAIN":[...],
*       "DATA_ELEMENT":[...],"TABLE":[...],"STRUCTURE":[...],
*       "TABLE_TYPE":[...]}, ...]} shape from ZDCCRT_DDIC_OBJ, scoped to
*       GT_PACK_NAMES. Each row's DETAIL_JSON is already the exact raw
*       JSON object ZLLD_PACKAGE_EXTRACTOR staged - never parsed, copied
*       through verbatim, same rule as BUILD_DEPENDENCIES_JSON's
*       SIGNATURE_JSON - so this only needs to bucket rows by DDIC_TYPE
*       and join them back into arrays.
*&---------------------------------------------------------------------*
FORM build_ddic_json_section CHANGING cv_ddic_json TYPE string.
  DATA: lr_pack        TYPE tt_devclass_range,
        lt_ddic        TYPE STANDARD TABLE OF zdccrt_ddic_obj WITH EMPTY KEY,
        lt_pkg_entries TYPE string_t,
        lt_domain      TYPE string_t,
        lt_dtel        TYPE string_t,
        lt_tabl        TYPE string_t,
        lt_stru        TYPE string_t,
        lt_ttyp        TYPE string_t,
        lv_pkg_json    TYPE string.

  CLEAR cv_ddic_json.
  PERFORM gt_pack_names_range CHANGING lr_pack.

  SELECT * FROM zdccrt_ddic_obj INTO TABLE @lt_ddic
    WHERE zpackage IN @lr_pack
    ORDER BY zpackage, ddic_type, ddic_name.

  LOOP AT gt_pack_names INTO DATA(lv_pack).
    CLEAR: lt_domain, lt_dtel, lt_tabl, lt_stru, lt_ttyp.

    LOOP AT lt_ddic INTO DATA(ls_ddic) WHERE zpackage = lv_pack.
      CASE ls_ddic-ddic_type.
        WHEN 'DOMA'. APPEND ls_ddic-detail_json TO lt_domain.
        WHEN 'DTEL'. APPEND ls_ddic-detail_json TO lt_dtel.
        WHEN 'TABL'. APPEND ls_ddic-detail_json TO lt_tabl.
        WHEN 'STRU'. APPEND ls_ddic-detail_json TO lt_stru.
        WHEN 'TTYP'. APPEND ls_ddic-detail_json TO lt_ttyp.
      ENDCASE.
    ENDLOOP.

    lv_pkg_json = |\{"PACKAGE":"{ lv_pack }",| &&
                  |"DOMAIN":[{ concat_lines_of( table = lt_domain sep = `,` ) }],| &&
                  |"DATA_ELEMENT":[{ concat_lines_of( table = lt_dtel sep = `,` ) }],| &&
                  |"TABLE":[{ concat_lines_of( table = lt_tabl sep = `,` ) }],| &&
                  |"STRUCTURE":[{ concat_lines_of( table = lt_stru sep = `,` ) }],| &&
                  |"TABLE_TYPE":[{ concat_lines_of( table = lt_ttyp sep = `,` ) }]\}|.

    APPEND lv_pkg_json TO lt_pkg_entries.
  ENDLOOP.

  cv_ddic_json = |\{"PACKAGES":[{ concat_lines_of( table = lt_pkg_entries sep = `,` ) }]\}|.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  EXPORT_CANDIDATES
*&---------------------------------------------------------------------*
FORM export_candidates.
  DATA: lt_blocks       TYPE string_t,
        lv_metadata     TYPE string,
        lv_header       TYPE string,
        lv_package      TYPE devclass,
        lv_source       TYPE string,
        lv_dep_json     TYPE string,
        lv_ddic_json    TYPE string,
        lv_block        TYPE string,
        lt_top_names    TYPE string_t,
        lv_proceed      TYPE abap_bool.

  CLEAR lt_top_names.
  LOOP AT gt_adjusted_sorted INTO DATA(lv_name) TO p_topn.
    APPEND lv_name TO lt_top_names.
  ENDLOOP.

  PERFORM build_metadata_block CHANGING lv_metadata.

  IF lt_top_names IS NOT INITIAL.
    READ TABLE gt_obj_score INTO DATA(ls_first) WITH KEY object_name = lt_top_names[ 1 ].
    lv_package = ls_first-package.
  ELSE.
    lv_package = 'UNKNOWN'.
  ENDIF.

  PERFORM build_ddic_json_section CHANGING lv_ddic_json.

  lv_header = |RUN_TYPE: INCREMENTAL\n| &&
              |PACKAGE: { lv_package }\n| &&
              |EXTRACTED_AT: { sy-datum DATE = ISO } { sy-uzeit TIME = ISO }\n| &&
              |REMOVED_OBJECTS: NONE\n| &&
              |\n| &&
              |--- DDIC ---\n| &&
              |{ lv_ddic_json }|.

  APPEND lv_metadata TO lt_blocks.
  APPEND lv_header TO lt_blocks.

  LOOP AT lt_top_names INTO lv_name.
    READ TABLE gt_obj_score INTO DATA(ls_cand) WITH KEY object_name = lv_name.
    CHECK sy-subrc = 0.

    PERFORM build_source_text USING lv_name CHANGING lv_source.
    PERFORM build_dependencies_json USING lv_name ls_cand-type CHANGING lv_dep_json.

    lv_block = |=== OBJECT: { lv_name } ===\n| &&
               |TYPE: { ls_cand-type }\n| &&
               |PACKAGE: { ls_cand-package }\n| &&
               |FINAL_SCORE: { ls_cand-final_score DECIMALS = 4 }\n| &&
               |SEMANTIC_SCORE: { ls_cand-semantic DECIMALS = 4 }\n| &&
               |STRUCTURAL_SCORE: { ls_cand-structural DECIMALS = 4 }\n| &&
               |TIER_ADJUSTMENT_APPLIED: { ls_cand-tier_label }\n| &&
               |\n| &&
               |--- SOURCE ---\n| &&
               |{ lv_source }\n| &&
               |\n| &&
               |--- DEPENDENCIES ---\n| &&
               |{ lv_dep_json }\n| &&
               |=== END OBJECT ===|.

    APPEND lv_block TO lt_blocks.
  ENDLOOP.

  gv_filename = |LLD_CANDIDATES_{ concat_lines_of( table = gt_pack_names sep = `_` ) }.TXT|.
  PERFORM build_full_path.

  PERFORM confirm_overwrite_if_exists CHANGING lv_proceed.
  IF lv_proceed = abap_false.
    WRITE: / 'Export cancelled - existing file was not overwritten.'.
    RETURN.
  ENDIF.

  DATA(lv_full_content) = concat_lines_of( table = lt_blocks sep = |\n\n| ).
  PERFORM download_output_file USING lv_full_content.

  WRITE: / 'Candidate file written to:', gv_full_path.
  WRITE: / 'Candidates exported:', lines( lt_top_names ).
  WRITE: / 'Likely new object:', gv_likely_new_object.
  IF gv_new_object_reason IS NOT INITIAL.
    WRITE: / 'Reason:', gv_new_object_reason.
  ENDIF.
  WRITE: / 'Disagreements:', lines( gt_disagreements ).
  LOOP AT gt_disagreements INTO DATA(ls_dis).
    WRITE: / '  -', ls_dis-kind, ':', ls_dis-message.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_METADATA_BLOCK
*&---------------------------------------------------------------------*
FORM build_metadata_block CHANGING cv_metadata TYPE string.
  DATA lv_lines TYPE string_t.

  APPEND '=== RETRIEVAL METADATA ===' TO lv_lines.
  APPEND |REQUIREMENT_FILE: { p_reqf }| TO lv_lines.
  APPEND |REQUIREMENT: { gv_requirement_text }| TO lv_lines.
  APPEND |PACKAGES: { concat_lines_of( table = gt_pack_names sep = `, ` ) }| TO lv_lines.
  APPEND |SCOPING_NOTE: { COND #( WHEN p_note IS NOT INITIAL THEN p_note ELSE '(none)' ) }| TO lv_lines.
  APPEND |CONFIDENCE: { COND #( WHEN p_conf IS NOT INITIAL THEN p_conf ELSE '(none)' ) }| TO lv_lines.
  APPEND |LIKELY_NEW_OBJECT: { gv_likely_new_object }| TO lv_lines.
  APPEND |LIKELY_NEW_OBJECT_REASON: { COND #( WHEN gv_new_object_reason IS NOT INITIAL THEN gv_new_object_reason ELSE '(none)' ) }| TO lv_lines.
  APPEND |DISAGREEMENTS: { lines( gt_disagreements ) }| TO lv_lines.
  LOOP AT gt_disagreements INTO DATA(ls_dis).
    APPEND |  - [{ ls_dis-kind }] { ls_dis-message }| TO lv_lines.
  ENDLOOP.
  APPEND '=== END RETRIEVAL METADATA ===' TO lv_lines.

  cv_metadata = concat_lines_of( table = lv_lines sep = |\n| ).
ENDFORM.

*&=======================================================================*
*& Requirement file input (P_REQF points at a local .txt file holding
*& the requirement text, instead of typing it into a single-line
*& selection-screen field - a dynpro PARAMETER has a hard-limited
*& visible/enterable width even when its ABAP type is STRING, so a
*& file is the only way to give this an effectively unbounded length)
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  F4_REQUIREMENT_FILE
*&---------------------------------------------------------------------*
*       File picker (not the folder picker P_FILE/F4_OUTPUT_FOLDER
*       uses below) - cl_gui_frontend_services=>file_open_dialog is the
*       standard counterpart to directory_browse.
*&---------------------------------------------------------------------*
FORM f4_requirement_file CHANGING cv_file TYPE string.
  DATA: lt_files       TYPE filetable,
        lv_rc          TYPE i,
        ls_file        TYPE file_table.

  cl_gui_frontend_services=>file_open_dialog(
    EXPORTING
      window_title      = 'Select Requirement Text File'
      default_extension  = 'txt'
      file_filter        = 'Text Files (*.txt)|*.txt|All Files (*.*)|*.*'
      initial_directory  = 'C:\'
    CHANGING
      file_table         = lt_files
      rc                 = lv_rc
    EXCEPTIONS
      file_open_dialog_failed = 1
      cntl_error               = 2
      error_no_gui              = 3
      not_supported_by_gui      = 4
      OTHERS                    = 5 ).

  CHECK sy-subrc = 0 AND lv_rc > 0.

  READ TABLE lt_files INTO ls_file INDEX 1.
  IF sy-subrc = 0.
    cv_file = ls_file-filename.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  LOAD_REQUIREMENT_TEXT
*&---------------------------------------------------------------------*
*       Reads P_REQF via GUI_UPLOAD and joins its lines (space-
*       separated - the requirement is tokenized as one blob of text
*       downstream, so line breaks in the source file carry no meaning
*       here) into GV_REQUIREMENT_TEXT, which every downstream form
*       reads instead of a selection-screen field directly.
*&---------------------------------------------------------------------*
FORM load_requirement_text.
  TYPES: BEGIN OF ts_line,
           line TYPE string,
         END OF ts_line.
  DATA lt_lines TYPE STANDARD TABLE OF ts_line WITH EMPTY KEY.

  CLEAR: gv_requirement_text, gv_error.

  " CODEPAGE = '4110' (UTF-8) - without it, GUI_UPLOAD's default ASC-mode
  " read assumes a single-byte codepage, which corrupts multi-byte UTF-8
  " characters (smart quotes, em-dashes, etc. commonly present in
  " requirement text pasted from Word/PDF) into mojibake.
  CALL FUNCTION 'GUI_UPLOAD'
    EXPORTING
      filename                = p_reqf
      filetype                = 'ASC'
      codepage                = '4110'
    TABLES
      data_tab                = lt_lines
    EXCEPTIONS
      file_open_error         = 1
      file_read_error         = 2
      no_batch                = 3
      gui_refuse_filetransfer = 4
      invalid_type            = 5
      no_authority            = 6
      unknown_error           = 7
      bad_data_format         = 8
      header_not_allowed      = 9
      separator_not_allowed   = 10
      header_too_long         = 11
      unknown_dp_error        = 12
      access_denied           = 13
      dp_out_of_memory        = 14
      disk_full               = 15
      dp_timeout              = 16
      OTHERS                  = 17.

  IF sy-subrc <> 0.
    MESSAGE |Could not read requirement file '{ p_reqf }'.| TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
    RETURN.
  ENDIF.

  LOOP AT lt_lines INTO DATA(ls_line).
    gv_requirement_text = COND #( WHEN gv_requirement_text IS INITIAL THEN ls_line-line
                                   ELSE gv_requirement_text && ' ' && ls_line-line ).
  ENDLOOP.

  IF gv_requirement_text IS INITIAL.
    MESSAGE |Requirement file '{ p_reqf }' is empty.| TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
  ENDIF.
ENDFORM.

*&=======================================================================*
*& File writer (same GUI_DOWNLOAD/folder-picker pattern used elsewhere
*& in this toolset - near-duplicate of ZLLD_PACKAGE_EXTRACTOR's own
*& F4_OUTPUT_FOLDER/BUILD_FULL_PATH/CONFIRM_OVERWRITE_IF_EXISTS/
*& DOWNLOAD_OUTPUT_FILE, since the two are separate REPORTs with no
*& shared include)
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  F4_OUTPUT_FOLDER
*&---------------------------------------------------------------------*
FORM f4_output_folder CHANGING cv_file TYPE string.
  DATA lv_selected_folder TYPE string.

  cl_gui_frontend_services=>directory_browse(
    EXPORTING
      window_title          = 'Select Output Folder'
      initial_folder        = 'C:\'
    CHANGING
      selected_folder       = lv_selected_folder
    EXCEPTIONS
      cntl_error            = 1
      error_no_gui          = 2
      not_supported_by_gui  = 3
      OTHERS                = 4 ).

  IF sy-subrc = 0 AND lv_selected_folder IS NOT INITIAL.
    cv_file = lv_selected_folder.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_FULL_PATH
*&---------------------------------------------------------------------*
FORM build_full_path.
  gv_full_path = p_file.
  IF substring( val = gv_full_path off = strlen( gv_full_path ) - 1 ) <> '\'.
    gv_full_path = gv_full_path && '\'.
  ENDIF.
  gv_full_path = gv_full_path && gv_filename.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  CONFIRM_OVERWRITE_IF_EXISTS
*&---------------------------------------------------------------------*
FORM confirm_overwrite_if_exists CHANGING cv_proceed TYPE abap_bool.
  DATA: lv_exists TYPE abap_bool,
        lv_answer TYPE c LENGTH 1.

  cv_proceed = abap_true.

  cl_gui_frontend_services=>file_exist(
    EXPORTING
      file   = gv_full_path
    RECEIVING
      result = lv_exists
    EXCEPTIONS
      OTHERS = 1 ).

  CHECK lv_exists = abap_true.

  CALL FUNCTION 'POPUP_TO_CONFIRM'
    EXPORTING
      titlebar       = 'File Already Exists'
      text_question  = |{ gv_full_path } already exists. Overwrite it?|
      text_button_1  = 'Yes'
      text_button_2  = 'No'
      default_button = '2'
    IMPORTING
      answer         = lv_answer
    EXCEPTIONS
      text_not_found = 1
      OTHERS         = 2.

  IF lv_answer <> '1'.
    cv_proceed = abap_false.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  DOWNLOAD_OUTPUT_FILE
*&---------------------------------------------------------------------*
*       IV_CONTENT is the whole export text, one string - split on
*       newline into GUI_DOWNLOAD's line-table shape (the same shape
*       ZLLD_PACKAGE_EXTRACTOR's own GT_OUTPUT already uses).
*&---------------------------------------------------------------------*
FORM download_output_file USING iv_content TYPE string.
  TYPES: BEGIN OF ts_line, line TYPE string, END OF ts_line.
  DATA lt_output TYPE STANDARD TABLE OF ts_line WITH EMPTY KEY.
  DATA lt_raw TYPE string_t.

  SPLIT iv_content AT cl_abap_char_utilities=>newline INTO TABLE lt_raw.
  LOOP AT lt_raw INTO DATA(lv_line).
    APPEND VALUE #( line = lv_line ) TO lt_output.
  ENDLOOP.

  " CODEPAGE = '4110' (UTF-8), matching LOAD_REQUIREMENT_TEXT's
  " GUI_UPLOAD - the requirement text (and source code comments/
  " literals) can carry non-ASCII characters (smart quotes, etc.);
  " without an explicit codepage GUI_DOWNLOAD's default ASC-mode write
  " uses a codepage that cannot represent them and substitutes garbage,
  " even though they were read correctly into memory on the way in.
  CALL FUNCTION 'GUI_DOWNLOAD'
    EXPORTING
      filename         = gv_full_path
      filetype         = 'ASC'
      codepage         = '4110'
    TABLES
      data_tab         = lt_output
    EXCEPTIONS
      file_write_error = 1
      OTHERS           = 2.

  IF sy-subrc <> 0.
    WRITE: / 'Error downloading file. Check if the file path is valid.'.
  ENDIF.
ENDFORM.
